import psutil
import time

# Detection module
from detection.threat_detector import (
    detect_cpu,
    detect_ram,
    detect_disk
)

# AI decision module
from ai.decision_engine import decide

# Response module
from response.response_engine import (
    respond_to_cpu,
    respond_to_ram,
    respond_to_disk
)

# Database module
from database.database_manager import (
    create_database,
    insert_log
)


print("======= ACTDRS System Monitor =======")

# Create database and table
create_database()


while True:

    # =========================
    # SYSTEM MONITORING
    # =========================

    cpu = psutil.cpu_percent(interval=1)
    memory = psutil.virtual_memory()
    disk = psutil.disk_usage('/')

    print("-----------------------")

    # =========================
    # CPU DETECTION
    # =========================

    cpu_status = detect_cpu(cpu)

    print(f"CPU Usage : {cpu}%")
    print(f"CPU Status : {cpu_status}")

    # =========================
    # RAM DETECTION
    # =========================

    ram_status = detect_ram(memory.percent)

    print(f"RAM Usage : {memory.percent}%")
    print(f"RAM Status : {ram_status}")

    # =========================
    # DISK DETECTION
    # =========================

    disk_status = detect_disk(disk.percent)

    print(f"Disk Usage : {disk.percent}%")
    print(f"Disk Status : {disk_status}")

    # =========================
    # AI DECISION
    # =========================

    decision = decide(
        cpu_status,
        ram_status,
        disk_status
    )

    print(f"\nAI DECISION : {decision}")

    # =========================
    # DATABASE LOGGING
    # =========================

    insert_log(
        cpu,
        memory.percent,
        disk.percent,
        decision
    )

    print("Log saved to database.")

    # =========================
    # RESPONSE ENGINE
    # =========================

    respond_to_cpu(cpu_status)
    respond_to_ram(ram_status)
    respond_to_disk(disk_status)

    # =========================
    # RUNNING PROCESSES
    # =========================

    print("\nRunning Processes\n")

    for process in psutil.process_iter(['pid', 'name']):
        print(process.info)

    print("\nWaiting for next scan...\n")
    print("--------------------------------")

    time.sleep(3)
