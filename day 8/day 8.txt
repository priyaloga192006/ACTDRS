from datetime import datetime

DB_NAME = "actdrs.db"


def create_database():

    conn = sqlite3.connect(DB_NAME)

    cursor = conn.cursor()

    cursor.execute("""
    CREATE TABLE IF NOT EXISTS logs(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        timestamp TEXT,
        cpu REAL,
        ram REAL,
        disk REAL,
        decision TEXT
    )
    """)

    conn.commit()
    conn.close()


def insert_log(cpu, ram, disk, decision):

    conn = sqlite3.connect(DB_NAME)

    cursor = conn.cursor()

    cursor.execute("""
    INSERT INTO logs(timestamp,cpu,ram,disk,decision)
    VALUES(?,?,?,?,?)
    """, (
        datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        cpu,
        ram,
        disk,
        decision
    ))

    conn.commit()
